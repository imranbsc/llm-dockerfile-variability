```dockerfile
# Use an official Node.js runtime as a parent image
FROM node:14

# Set the working directory in the container to /app
WORKDIR /app

# Copy the package.json and package-lock.json files from the local directory to the container at /app
COPY package*.json ./

# Install any needed packages specified in package.json
RUN npm install

# Copy the rest of the code from the local directory to the container at /app
COPY . .

# Define the command to run when the container launches
CMD ["node", "worker.js"]
```
