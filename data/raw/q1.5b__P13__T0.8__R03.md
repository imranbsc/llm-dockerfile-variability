```Dockerfile
# Use an official OpenJDK runtime as a parent image
FROM openjdk:11-jre-slim

# Set the working directory in the container to /app
WORKDIR /app

# Copy the current directory contents into the container at /app
COPY . /app

# Install any needed packages specified in requirements.txt
RUN apt-get update && apt-get install -y \
    && rm -rf /var/lib/apt/lists/*

# Install RabbitMQ
RUN wget https://www.rabbitmq.com/releases/rabbitmq-server/v3.9.15/rabbitmq-server-3.9.15-1-macosx.tar.xz && \
    tar -xvzf rabbitmq-server-3.9.15-1-macosx.tar.xz && \
    mv rabbitmq-server-3.9.15 /opt && \
    rm rabbitmq-server-3.9.15-1-macosx.tar.xz

# Run Spring Boot application
CMD ["java", "-jar", "your-spring-boot-application.jar"]
```
