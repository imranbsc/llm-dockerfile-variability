```Dockerfile
# Use an official OpenJDK runtime as a parent image
FROM openjdk:11-jdk-slim

# Set the working directory in the container to /app
WORKDIR /app

# Copy the current directory contents into the container at /app
COPY . /app

# Install any needed packages specified in requirements.txt
RUN apt-get update && apt-get install -y \
    && rm -rf /var/lib/apt/lists/*

# Set the environment variable to enable logging in the container
ENV JAVA_OPTS="-Dlogging.level.root=DEBUG"

# Define the command to run the Spring Batch job
CMD ["java", "-jar", "target/spring-batch-job-0.0.1-SNAPSHOT.jar"]
```
