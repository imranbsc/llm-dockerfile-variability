```dockerfile
# Use the official .NET Worker Service template as a base image
FROM mcr.microsoft.com/dotnet/worker:6.0 AS base

# Set the working directory in the container
WORKDIR /app

# Copy the csproj file and restore dependencies
COPY *.csproj ./app/
RUN dotnet restore

# Copy the rest of the project source code
COPY . ./app/

# Set the environment variables
ENV ASPNETCORE_ENVIRONMENT=Development

# Build the application
RUN dotnet publish --configuration Release --output publish

# Create a non-root user to run the application
RUN useradd -m dotnetuser

# Copy the publish directory into the container
COPY --from=base /app/publish /app/

# Switch to the non-root user
USER dotnetuser

# Expose the port used by the application
EXPOSE 5001

# Run the application
CMD ["dotnet", "run"]
```
