```Dockerfile
# Use a multi-stage build to minimize the final image size
FROM mcr.microsoft.com/dotnet/aspnet:7.0 AS base
WORKDIR /app

# Copy the application project files
COPY *.csproj ./
RUN dotnet restore

# Copy the application source code
COPY . .

# Build the application
RUN dotnet build -c Release

# Set the working directory for the background worker
WORKDIR /app/WorkerService

# Expose the port that the background worker will listen on
EXPOSE 5000

# Run the background worker
ENTRYPOINT ["dotnet", "WorkerService.dll", "Run"]
```
