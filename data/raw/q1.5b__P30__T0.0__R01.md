```Dockerfile
# Use a base image with .NET SDK
FROM mcr.microsoft.com/dotnet/sdk:6.0 AS build-env

# Set the working directory in the container
WORKDIR /app

# Copy the solution file and restore the dependencies
COPY *.sln ./
COPY *.csproj ./
RUN dotnet restore

# Copy the source code to the container
COPY . .

# Set the environment variable for Quartz.NET
ENV Quartz.NET_CONFIG_FILE=Quartz.config

# Build the application
RUN dotnet build -c Release

# Create a non-root user and group
RUN useradd -m -s /bin/bash user && groupadd -g user user

# Copy the application to the user's home directory
COPY --from=build-env /app/bin/Release/net6.0/user /home/user

# Set the user and group for the application
USER user

# Expose the Quartz.NET port
EXPOSE 5000

# Run the Quartz.NET application
CMD ["dotnet", "Quartz.ConsoleApp.dll"]
```
