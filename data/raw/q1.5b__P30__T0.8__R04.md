```Dockerfile
# Use an official .NET SDK runtime as a parent image
FROM mcr.microsoft.com/dotnet/sdk:5.0

# Set the working directory in the container
WORKDIR /app

# Copy the current directory contents into the container at /app
COPY . /app

# Install any needed packages specified in requirements.txt
RUN dotnet add package Quartz

# Create the Quartz scheduler instance
RUN dotnet Quartz.Host.CreateDefaultHostBuilder(Environment.GetEnvironmentVariables())
    .ConfigureServices((context, services) =>
    {
        // Register Quartz services
        services.AddQuartz()
            .AddInMemoryStorage()
            .AddQuartzLogging()
            .AddQuartzServer();
    })
    .Build()
    .Run();

# Expose the default HTTP port for the web server
EXPOSE 80

# Set the command to run when the container launches
CMD ["dotnet", "QuartzHost.dll"]
```
