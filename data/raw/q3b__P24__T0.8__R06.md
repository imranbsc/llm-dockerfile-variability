```Dockerfile
# Use an official PHP runtime as a parent image
FROM php:7.4-cli

# Set the working directory in the container
WORKDIR /usr/src/app

# Copy the current directory contents into the container at /usr/src/app
COPY . /usr/src/app

# Install any needed packages specified in requirements.txt
RUN apt-get update && apt-get install -y \
    libzip-dev \
    zip \
    git \
    && docker-php-ext-install zip \
    && composer install

# Make port 80 available to the world outside this container
EXPOSE 80

# Run app/console when the container launches
CMD ["php", "app/console"]
```

Note: This Dockerfile assumes that your PHP application has a Symfony Console command named `app/console`. If your command is different, you will need to adjust the `CMD` line accordingly.
