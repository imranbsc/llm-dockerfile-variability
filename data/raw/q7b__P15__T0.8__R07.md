```Dockerfile
# Use an official OpenJDK runtime as a parent image
FROM openjdk:11-jre-slim

# Set the maintainer label
LABEL maintainer="your-email@example.com"

# Set the working directory in the container
WORKDIR /app

# Copy the compiled Java application from the build directory
COPY target/your-application.jar /app/your-application.jar

# Set the environment variable for Spring Batch job scheduling
ENV SPRING_BATCH_JOB=your-job-name
ENV SPRING_BATCH_JOB_PARAMS="param1=value1,param2=value2"

# Expose the port on which the application will run (if needed)
# EXPOSE 8080

# Command to run the Java application
CMD ["java", "-jar", "your-application.jar"]
```
