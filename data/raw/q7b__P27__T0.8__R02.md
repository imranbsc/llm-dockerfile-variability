```dockerfile
# Use an official .NET SDK image as the base image
FROM mcr.microsoft.com/dotnet/sdk:6.0 AS build-env

# Set the working directory
WORKDIR /app

# Copy the project file and restore the dependencies
COPY *.csproj ./
RUN dotnet restore

# Copy the project source code and build the application
COPY . ./
RUN dotnet build -c Release -o out

# Use an official .NET runtime image for the production environment
FROM mcr.microsoft.com/dotnet/aspnet:6.0

# Create the destination directory
WORKDIR /app

# Copy the build output from the build-env stage
COPY --from=build-env /app/out .

# Expose the port the app runs on
EXPOSE 80

# Set the command to run the application
ENTRYPOINT ["dotnet", "YourApplicationName.dll"]
```

This Dockerfile sets up a multi-stage build process for a C# ASP.NET Core MVC application. It first builds the application using the SDK image, then copies the resulting output to a smaller ASP.NET Core runtime image to create a production-ready container.
