```dockerfile
# Use the official .NET SDK image as the base image
FROM mcr.microsoft.com/dotnet/sdk:6.0 AS build-env
WORKDIR /app

# Copy the project file and restore dependencies
COPY *.csproj ./
RUN dotnet restore

# Copy the project files and build the application
COPY . ./
RUN dotnet publish -c Release -o out

# Use the official .NET runtime image as the base image for the final image
FROM mcr.microsoft.com/dotnet/aspnet:6.0
WORKDIR /app

# Copy the published application from the build stage
COPY --from=build-env /app/out .

# Set the command to run the application
ENTRYPOINT ["dotnet", "YourProjectName.dll"]
```

Replace `YourProjectName.dll` with the actual name of your C# project DLL. This Dockerfile sets up a multi-stage build process where the application is built in a development environment and then published for a smaller, production-ready image.
