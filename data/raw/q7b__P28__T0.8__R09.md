```dockerfile
# Use the official .NET SDK image as the base image
FROM mcr.microsoft.com/dotnet/sdk:6.0 AS build-env

# Set the working directory in the container
WORKDIR /app

# Copy the project file and restore dependencies
COPY *.csproj ./
RUN dotnet restore

# Copy the rest of the project files and build the application
COPY . ./
RUN dotnet publish -c Release -o out

# Use the official .NET runtime image as the final image
FROM mcr.microsoft.com/dotnet/aspnet:6.0

# Set the working directory in the container
WORKDIR /app

# Copy the published application from the build stage
COPY --from=build-env /app/out .

# Run the application as a background worker
ENTRYPOINT ["dotnet", "YourWorkerServiceName.dll"]
```

Replace `YourWorkerServiceName.dll` with the actual name of your .NET Worker Service DLL. This Dockerfile sets up a multi-stage build process to first build the application and then create a lean runtime image that only includes the necessary files for running the worker service.
